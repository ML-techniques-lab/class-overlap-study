		\caption{Média Global: #1 (#2).}
	\end{figure}
}

\newcommand{\plotGrid}[3]{
	\begin{figure}[H]
		\centering
		\includegraphics[width=\textwidth]{plots/Granular_Grids/#1/#2/grid_#1_#2_ds_#3.pdf}
		\caption{Grid #1 (#2) - Datasets #3.}
	\end{figure}
}

\newcommand{\cellMedia}[3]{

		\includegraphics[width=\linewidth]{avg_lineplot_#2_#3.pdf}
		\caption{\scriptsize #2 (#3)} % Fonte menor para não poluir
	\end{subfigure}
}
